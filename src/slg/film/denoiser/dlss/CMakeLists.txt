cmake_minimum_required(VERSION 3.20)
project(SuperLuxCoreDLSS LANGUAGES C CXX)
set(DLSS_ROOT "" CACHE PATH "External official NVIDIA DLSS SDK root; never bundled implicitly")
if(NOT EXISTS "${DLSS_ROOT}/include/nvsdk_ngx.h")
  message(FATAL_ERROR "DLSS_ROOT must point to the official SDK containing include/nvsdk_ngx.h")
endif()
get_filename_component(SLC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_library(superluxcore_dlss SHARED runtime.cpp "${SLC_ROOT}/deps/cuew/src/cuew.cpp")
target_compile_features(superluxcore_dlss PRIVATE cxx_std_17)
target_include_directories(superluxcore_dlss PRIVATE "${DLSS_ROOT}/include" "${SLC_ROOT}/deps/cuew/include")
set_target_properties(superluxcore_dlss PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
target_link_libraries(superluxcore_dlss PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(superluxcore_dlss PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  # The header-only NVIDIA loader discovers the installed driver via registry.
  target_link_libraries(superluxcore_dlss PRIVATE advapi32)
endif()
if(MSVC)
  # NVIDIA's header-only loader expands variadic argument pairs recursively.
  target_compile_options(superluxcore_dlss PRIVATE /Zc:preprocessor)
endif()
install(TARGETS superluxcore_dlss LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(FILES "${DLSS_ROOT}/LICENSE.txt" DESTINATION . RENAME NVIDIA-DLSS-LICENSE.txt)
install(FILES "${SLC_ROOT}/deps/cuew/LICENSE" DESTINATION . RENAME CUEW-LICENSE.txt)
